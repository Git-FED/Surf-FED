electron/renderer.js — the split-view engine
js
// ===========================================================================
// Surf FED — 3-pane split view renderer
// ===========================================================================
// Design:
//   - All <webview> elements live in #webviewParking, never re-parented.
//   - Each visible pane's webview is positioned via absolute left/top/width/height
//     to overlay that pane's rect. This avoids the Electron reload-on-reparent trap.
//   - Panes are geometry + headers only. Webviews are layered under the headers.
// ===========================================================================

const WEBVIEW_PARTITION = 'persist:surf-fed';
const STORAGE_KEY = 'surf-fed-splitview';

const state = {
  layout: 1,                 // 1, 2, or 3
  panes: [null, null, null], // tabId per pane slot (retained even when hidden)
  focusedPane: 0,
  tabs: new Map(),           // tabId -> { id, webview, title, url, loadCount }
  nextId: 1,
};

// --- DOM refs ---------------------------------------------------------------
const $ = (id) => document.getElementById(id);
const paneEls    = [0,1,2].map(i => $('pane-' + i));
const dividerEls = [0,1].map(i => $('divider-' + i));
const paneArea   = $('paneArea');
const parking    = $('webviewParking');
const urlBar     = $('urlBar');
const tabStrip   = $('tabStrip');

// --- Webview creation -------------------------------------------------------
function createWebview(tabId, url) {
  const wv = document.createElement('webview');
  wv.setAttribute('partition', WEBVIEW_PARTITION);
  wv.setAttribute('src', url || 'about:blank');
  wv.setAttribute('allowpopups', '');
  wv.classList.add('tab-webview');
  wv.dataset.tabId = String(tabId);
  parking.appendChild(wv);
  return wv;
}

// --- Tab lifecycle ----------------------------------------------------------
function createTab(url) {
  const id = state.nextId++;
  const wv = createWebview(id, url);
  const tab = { id, webview: wv, title: 'New Tab', url: url || 'about:blank', loadCount: 0 };
  state.tabs.set(id, tab);

  wv.addEventListener('did-start-loading', () => { tab.loadCount++; });
  wv.addEventListener('page-title-updated', (e) => {
    tab.title = e.title || 'New Tab';
    renderTabStrip();
    renderPaneHeaders();
  });
  wv.addEventListener('did-navigate', (e) => {
    tab.url = e.url;
    if (state.panes[state.focusedPane] === id) urlBar.value = e.url;
  });
  wv.addEventListener('did-navigate-in-page', (e) => {
    tab.url = e.url;
    if (state.panes[state.focusedPane] === id) urlBar.value = e.url;
  });
  wv.addEventListener('focus', () => {
    const idx = state.panes.indexOf(id);
    if (idx >= 0 && idx < state.layout) focusPane(idx);
  });

  // Assign to focused pane. Displaced tab is parked (stays in state.tabs, webview stays alive).
  state.panes[state.focusedPane] = id;

  applyLayout();
  renderTabStrip();
  return tab;
}

function closeTab(tabId) {
  const tab = state.tabs.get(tabId);
  if (!tab) return;

  // Remove from pane assignment
  for (let i = 0; i < 3; i++) {
    if (state.panes[i] === tabId) state.panes[i] = null;
  }
  // Destroy webview
  tab.webview.remove();
  state.tabs.delete(tabId);

  // Refill the pane(s) that lost a tab from parked tabs
  refillPanesFromParked();

  // If nothing is open at all, create a fresh tab in the focused pane
  if (state.tabs.size === 0) {
    createTab('about:blank');
  }

  applyLayout();
  renderTabStrip();
}

// --- Parking helpers --------------------------------------------------------
function visibleTabIds() {
  return state.panes.slice(0, state.layout).filter(Boolean);
}
function isParked(tabId) {
  return !state.panes.includes(tabId);
}
function firstParkedTab() {
  for (const tab of state.tabs.values()) {
    if (isParked(tab.id)) return tab;
  }
  return null;
}
function refillPanesFromParked() {
  for (let i = 0; i < state.layout; i++) {
    if (!state.panes[i]) {
      const parked = firstParkedTab();
      if (parked) state.panes[i] = parked.id;
    }
  }
}

// --- Layout -----------------------------------------------------------------
function setLayout(n) {
  if (![1,2,3].includes(n)) return;
  state.layout = n;

  // Growing: fill any new visible panes from parked tabs.
  for (let i = 0; i < n; i++) {
    if (!state.panes[i]) {
      const parked = firstParkedTab();
      if (parked) state.panes[i] = parked.id;
    }
  }

  // Shrinking: panes beyond `n` keep their assignment but become hidden.
  // Their webviews stay alive in parking (visibility hidden).

  // Clamp focus into visible range.
  if (state.focusedPane >= n) state.focusedPane = n - 1;

  applyLayout();
  renderTabStrip();
}

function focusPane(index) {
  if (index < 0 || index >= state.layout) return;
  state.focusedPane = index;
  applyLayout();
  const tabId = state.panes[index];
  if (tabId) {
    const tab = state.tabs.get(tabId);
    if (tab) urlBar.value = tab.url || '';
  }
}

function swapPanes(a, b) {
  if (a === b) return;
  if (a < 0 || a >= 3 || b < 0 || b >= 3) return;
  const tmp = state.panes[a];
  state.panes[a] = state.panes[b];
  state.panes[b] = tmp;
  applyLayout();
  renderTabStrip();
}

// --- DOM render -------------------------------------------------------------
function applyLayout() {
  // Panes: visible for i < layout
  for (let i = 0; i < 3; i++) {
    paneEls[i].classList.toggle('visible', i < state.layout);
    paneEls[i].classList.toggle('focused', i === state.focusedPane && i < state.layout);
  }
  // Dividers: between visible panes
  dividerEls[0].classList.toggle('visible', state.layout >= 2);
  dividerEls[1].classList.toggle('visible', state.layout >= 3);

  layoutWebviews();
  renderPaneHeaders();
  syncLayoutButtons();
  saveState();
}

function layoutWebviews() {
  const areaR = paneArea.getBoundingClientRect();

  for (const tab of state.tabs.values()) {
    const idx = state.panes.indexOf(tab.id);
    const visible = idx >= 0 && idx < state.layout;
    if (visible) {
      const r = paneEls[idx].getBoundingClientRect();
      tab.webview.style.left   = (r.left - areaR.left) + 'px';
      tab.webview.style.top    = (r.top  - areaR.top)  + 'px';
      tab.webview.style.width  = r.width  + 'px';
      tab.webview.style.height = r.height + 'px';
      tab.webview.style.visibility = 'visible';
      tab.webview.style.pointerEvents = 'auto';
      tab.webview.style.zIndex = '1';
    } else {
      tab.webview.style.visibility = 'hidden';
      tab.webview.style.pointerEvents = 'none';
    }
  }
}

function renderPaneHeaders() {
  for (let i = 0; i < 3; i++) {
    const header = paneEls[i].querySelector('.pane-title');
    const tabId = state.panes[i];
    const tab = tabId ? state.tabs.get(tabId) : null;
    header.textContent = tab ? tab.title : 'Empty';
  }
}

function renderTabStrip() {
  tabStrip.innerHTML = '';
  const activeIds = new Set(state.panes.slice(0, state.layout));
  for (const tab of state.tabs.values()) {
    const chip = document.createElement('div');
    chip.className = 'tab-chip' + (activeIds.has(tab.id) ? ' active' : '');
    chip.dataset.tabId = String(tab.id);
    chip.innerHTML = `<span>${escapeHtml(tab.title)}</span><button class="close">×</button>`;
    chip.addEventListener('click', (e) => {
      if (e.target.classList.contains('close')) return;
      // Clicking a chip moves it into the focused pane (parking the displaced tab).
      state.panes[state.focusedPane] = tab.id;
      applyLayout();
      renderTabStrip();
    });
    chip.querySelector('.close').addEventListener('click', (e) => {
      e.stopPropagation();
      closeTab(tab.id);
    });
    tabStrip.appendChild(chip);
  }
}

function syncLayoutButtons() {
  document.querySelectorAll('.layout-btn').forEach(btn => {
    btn.classList.toggle('active', Number(btn.dataset.layout) === state.layout);
  });
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}

// --- Persistence ------------------------------------------------------------
function saveState() {
  try {
    const snapshot = {
      layout: state.layout,
      panes: state.panes,
      focusedPane: state.focusedPane,
      tabs: [...state.tabs.values()].map(t => ({ id: t.id, url: t.url, title: t.title })),
      nextId: state.nextId,
    };

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return false;
    const snap = JSON.parse(raw);
    // Recreate tabs without loading them yet — we restore URLs on demand.
    state.nextId = snap.nextId || 1;
    for (const t of (snap.tabs || [])) {
      const id = t.id;
      const wv = createWebview(id, t.url);
      const tab = { id, webview: wv, title: t.title || 'New Tab', url: t.url, loadCount: 0 };
      state.tabs.set(id, tab);
      wv.addEventListener('page-title-updated', (e) => {
        tab.title = e.title || 'New Tab';
        renderTabStrip(); renderPaneHeaders();
      });
      wv.addEventListener('did-navigate', (e) => {
        tab.url = e.url;
        if (state.panes[state.focusedPane] === id) urlBar.value = e.url;
      });
    }
    state.layout = snap.layout || 1;
    state.panes = snap.panes || [null, null, null];
    state.focusedPane = snap.focusedPane || 0;
    applyLayout();
    renderTabStrip();
    return true;
  } catch (e) { return false; }
}

How the feature works (so any AI building it knows what "done" means)
Default behavior: every new tab/webview is muted on creation. webview.setAudioMuted(true) fires before any page load.

Whitelist: a per-origin allowlist stored in localStorage under surf-fed-audio-whitelist. On did-navigate, the controller checks new URL(url).origin against the list. If present, setAudioMuted(false). If not, stays muted.
